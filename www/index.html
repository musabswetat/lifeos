// --- محرك الإشعارات المعزز للخلفية والمنبهات الدقيقة --- function getNativePlugin() { if (window.Capacitor && window.Capacitor.Plugins && window.Capacitor.Plugins.LocalNotifications) { return window.Capacitor.Plugins.LocalNotifications; } return null; } // إنشاء وتجهيز قنوات الصوت والأذونات بشكل متزامن async function initNativeNotifications() { const plugin = getNativePlugin(); if (!plugin) return; try { // 1. طلب الأذونات الصريحة لنظام أندرويد const perm = await plugin.requestPermissions(); // 2. إنشاء القناة المخصصة لنغمة الماريمبا بأعلى أولوية (Importance: 5) await plugin.createChannel({ id: 'lifeos_marimba_channel_v5', name: 'تنبيهات Life OS المخصصة', description: 'قناة التنبيهات بصوت ماريمبا في الخلفية', importance: 5, visibility: 1, sound: 'marimba', vibration: true, lights: true }); // 3. إنشاء قناة احتياطية بالصوت الافتراضي لضمان عدم تفويت أي تنبيه await plugin.createChannel({ id: 'lifeos_fallback_channel', name: 'تنبيهات Life OS العامة', description: 'قناة التنبيهات العامة الاحتياطية', importance: 5, visibility: 1, sound: 'default', vibration: true }); // 4. تسجيل زر إيقاف التنبيه من شريط الإشعارات await plugin.registerActionTypes({ types: [{ id: 'ALARM_ACTIONS', actions: [{ id: 'dismiss_alarm', title: '🔕 إيقاف التنبيه', destructive: true }] }] }); plugin.addListener('localNotificationActionPerformed', () => { dismissActiveAlarm(); }); } catch (e) { console.error("فشل تهيئة قنوات الإشعارات:", e); } } initNativeNotifications(); // دالة الجدولة الصارمة للإيقاظ من وضع الخمول async function scheduleSystemNotification(title, minutes, reminderId) { const plugin = getNativePlugin(); if (!plugin) return; const notifTitle = title ? `⏰ تنبيه: ${title}` : `⏰ موعد التنبيه (${minutes} دقيقة)!`; const notifBody = title ? `حان الآن موعد: ${title}` : `انتهت مدة التنبيه في Life OS`; const targetDate = new Date(Date.now() + minutes * 60 * 1000); const safeId = Math.abs(Number(reminderId) % 2147483647); try { // المحاولة بالقناة المخصصة await plugin.schedule({ notifications: [{ title: notifTitle, body: notifBody, id: safeId, schedule: { at: targetDate, allowWhileIdle: true }, sound: 'marimba', actionTypeId: 'ALARM_ACTIONS', channelId: 'lifeos_marimba_channel_v5' }] }); } catch (err) { // Fallback التلقائي في حال فشل الصوت المخصص try { await plugin.schedule({ notifications: [{ title: notifTitle, body: notifBody, id: safeId, schedule: { at: targetDate, allowWhileIdle: true }, sound: 'default', actionTypeId: 'ALARM_ACTIONS', channelId: 'lifeos_fallback_channel' }] }); } catch (fallbackErr) { alert('تعذر جدولة الإشعار في نظام أندرويد. تأكد من تفعيل إذن الإشعارات للتطبيق.'); } } }
